### 🎨 二、 使用其他工具“填充肉身”的操作指南

#### 1. 填充动态汉字与大数字（字体转换工具）
代码中使用的标准字体无法显示“成洛大道”这类中文字符，时速 0 也过于纤细。
 * **推荐工具**：使用 **LVGL Online Font Converter** 或 PC 端的本地转换工具。
 * **填充方法**：
   1. 找一个精美的车载矢量字体（如 *思源黑体*、*Roboto Bold*）。
   2. 生成**小字号字体**（如 16px），范围输入 0123456789米成洛大道（或完整常用汉字表），导出并在代码中通过 LV_FONT_DECLARE() 声明，绑定到 lbl_road_name 和 lbl_distance 的 style。
   3. 生成一个**超大字号字体**（如 60px 或 80px），仅包含数字 0-9，专门绑定给中间的 lbl_speed_num（大时速）。
#### 2. 替换左转弯箭头与透视线（切图工具）
原图中带有透视角度的灰色车道线和精美的白色左转弯剪头，在微控制器（MCU）中用代码绘制极度消耗算力，**最完美的选择是直接贴图**。
 * **推荐工具**：Figma / Photoshop + **LVGL Image Converter**。
 * **填充方法**：
   1. 在设计软件中，绘制出纯黑背景加上两侧灰色透视线的背景图（尺寸 100x240），直接作为 nav_panel 的背景图贴上去。
   2. 单独绘制一个白色左转弯弯头的 PNG 图片（如 48x48 像素），通过 LVGL 图片转换工具转为 .c 数组。
   3. 在框架代码中，将 img_nav_arrow = lv_label_create(...) 变更为：
     ```c
     LV_IMG_DECLARE(my_turn_left_arrow_png); // 声明你转换后的图片数组
     img_nav_arrow = lv_img_create(nav_panel);
     lv_img_set_src(img_nav_arrow, &my_turn_left_arrow_png);
     
     ```
#### 3. 车道指示器的精细化控制（Flex 布局填充）
代码中已经使用 LVGL 的 **Flexbox（弹性盒子布局）** 将顶部的蓝色条子做好了自动对齐。
 * **填充方法**：目前代码中用的是 ↑ 字符占位。你可以单独做两个小图片资产：一个是白色直行箭头.png，一个是暗蓝色/灰色直行箭头.png（代表不可走车道）。在逻辑层根据高德/百度地图传来的车道数据，动态 lv_img_set_src() 切换各个车道小控件的图片，就能实现原图中“前几条车道亮、最后两条车道暗”的高级动态效果。

完整 UI 框架源码 (hud_ui_framework.c
/**
 * @file hud_ui_framework.c
 * @brief 专为 2.8寸 320x240 TFT 屏幕优化的车载 HUD 导航 UI 框架
 * @note 兼容 LVGL v8.x 和 v9.x 版本
 * 
 * 本框架将屏幕分为“左导航、中时速、右状态”三个黄金比例区域，
 * 并提供了完整的布局结构、样式设置以及运行时动态更新数据的 API 接口。
 */

#include "lvgl.h"

/* --- UI 控件对象句柄 --- */
static lv_obj_t * scr;

/* 三大核心区域容器 */
static lv_obj_t * nav_panel;    /* 左侧：路名、导航箭头、剩余距离 */
static lv_obj_t * speed_panel;  /* 中间：车道指示、大字时速 */
static lv_obj_t * status_panel; /* 右侧：限速标志、连接状态 */

/* 动态更新组件句柄 */
static lv_obj_t * lbl_road_name;   /* “成洛大道” 文本框 */
static lv_obj_t * lbl_distance;    /* “725米” 文本框 */
static lv_obj_t * img_nav_arrow;   /* 导航方向箭头（当前为字符占位，可填充为图片） */
static lv_obj_t * lbl_speed_num;   /* 大字时速数字 “0” */
static lv_obj_t * lane_container;  /* 顶部车道指示栏容器 */
static lv_obj_t * lbl_speed_limit; /* 红圈内部的限速数字 */
static lv_obj_t * img_status_link; /* 蓝牙/网络连接状态图标 */

/* --- UI 样式定义（CSS-Like Styles） --- */
static lv_style_t style_scr;
static lv_style_t style_panel_transparent;
static lv_style_t style_text_white;
static lv_style_t style_speed_large;
static lv_style_t style_lane_bar;
static lv_style_t style_speed_limit_circle;

/**
 * @brief 初始化 UI 样式，确保全屏渲染高效且统一
 */
static void init_ui_styles(void) {
    /* 主屏幕底色样式（全黑背景，如同原图） */
    lv_style_init(&style_scr);
    lv_style_set_bg_color(&style_scr, lv_color_black());
    lv_style_set_text_color(&style_scr, lv_color_white());

    /* 透明基础面板样式（用于划分左中右区域，无边框，无内边距） */
    lv_style_init(&style_panel_transparent);
    lv_style_set_bg_opa(&style_panel_transparent, LV_OPA_TRANSP);
    lv_style_set_border_width(&style_panel_transparent, 0);
    lv_style_set_pad_all(&style_panel_transparent, 0);

    /* 标准白色文本样式 */
    lv_style_init(&style_text_white);
    lv_style_set_text_color(&style_text_white, lv_color_white());
    lv_style_set_text_align(&style_text_white, LV_TEXT_ALIGN_CENTER);

    /* 核心大字时速样式 */
    lv_style_init(&style_speed_large);
    lv_style_set_text_color(&style_speed_large, lv_color_white());
    lv_style_set_text_align(&style_speed_large, LV_TEXT_ALIGN_CENTER);
    /* 实际开发中，请使用字体转换工具将此样式绑定至 60px~80px 的大数字字体 */

    /* 顶部车道深蓝色背景条样式 */
    lv_style_init(&style_lane_bar);
    lv_style_set_bg_color(&style_lane_bar, lv_color_make(24, 116, 205)); /* 天蓝色 */
    lv_style_set_bg_opa(&style_lane_bar, LV_OPA_COVER);
    lv_style_set_radius(&style_lane_bar, 4);
    lv_style_set_pad_all(&style_lane_bar, 2);

    /* 限速标志牌样式（白底、红圈、黑字） */
    lv_style_init(&style_speed_limit_circle);
    lv_style_set_bg_color(&style_speed_limit_circle, lv_color_white());
    lv_style_set_bg_opa(&style_speed_limit_circle, LV_OPA_COVER);
    lv_style_set_border_color(&style_speed_limit_circle, lv_color_make(220, 20, 60)); /* 胭脂红 */
    lv_style_set_border_width(&style_speed_limit_circle, 3);
    lv_style_set_radius(&style_speed_limit_circle, LV_RADIUS_CIRCLE); /* 完美正圆 */
    lv_style_set_text_color(&style_speed_limit_circle, lv_color_black());
}

/**
 * @brief 创建左侧导航面板 (路名、导航箭头、距离)
 */
static void create_left_navigation_panel(void) {
    nav_panel = lv_obj_create(scr);
    lv_obj_add_style(nav_panel, &style_panel_transparent, 0);
    lv_obj_set_size(nav_panel, 100, 240);
    lv_obj_align(nav_panel, LV_ALIGN_LEFT_MID, 5, 0);

    /* 1. 路名标签 ("成洛大道") */
    lbl_road_name = lv_label_create(nav_panel);
    lv_obj_add_style(lbl_road_name, &style_text_white, 0);
    lv_label_set_text(lbl_road_name, "成洛大道");
    /* 提示：外部工具生成的包含中文字体的 font 需绑定在此处 */
    lv_obj_align(lbl_road_name, LV_ALIGN_TOP_MID, 0, 15);

    /* 2. 导航箭头 */
    /* 建议后续替换为 lv_img_create，用外部工具导出的精美 PNG 箭头数组 */
    img_nav_arrow = lv_label_create(nav_panel);
    lv_obj_add_style(img_nav_arrow, &style_text_white, 0);
    lv_label_set_text(img_nav_arrow, "▲"); 
    lv_obj_align(img_nav_arrow, LV_ALIGN_CENTER, 0, 0);
    
    /* 3. 剩余距离标签 ("725米") */
    lbl_distance = lv_label_create(nav_panel);
    lv_obj_add_style(lbl_distance, &style_text_white, 0);
    lv_label_set_text(lbl_distance, "725米");
    lv_obj_align(lbl_distance, LV_ALIGN_BOTTOM_MID, 0, -20);
}

/**
 * @brief 创建中间时速面板 (顶部多车道条、大数字车速)
 */
static void create_center_speed_panel(void) {
    speed_panel = lv_obj_create(scr);
    lv_obj_add_style(speed_panel, &style_panel_transparent, 0);
    lv_obj_set_size(speed_panel, 140, 240);
    lv_obj_align(speed_panel, LV_ALIGN_CENTER, 0, 0);

    /* 1. 顶部车道指引外壳 */
    lane_container = lv_obj_create(speed_panel);
    lv_obj_add_style(lane_container, &style_lane_bar, 0);
    lv_obj_set_size(lane_container, 130, 28);
    lv_obj_align(lane_container, LV_ALIGN_TOP_MID, 0, 15);
    
    /* 使用 Flexbox 弹性布局，内部小箭头自动水平均匀排列 */
    lv_obj_set_layout(lane_container, LV_LAYOUT_FLEX);
    lv_obj_set_flex_flow(lane_container, LV_FLEX_FLOW_ROW);
    lv_obj_set_flex_align(lane_container, LV_FLEX_ALIGN_SPACE_EVENLY, LV_FLEX_ALIGN_CENTER, LV_FLEX_ALIGN_CENTER);

 /* 动态添加4个车道占位符 (↑ ↑ ↑ ↑) */
    for(int i = 0; i < 4; i++) {
        lv_obj_t * lane_arrow = lv_label_create(lane_container);
        lv_label_set_text(lane_arrow, "↑");
        lv_obj_add_style(lane_arrow, &style_text_white, 0);
        /* 实际业务中，你可以给不可通行的车道单独降低透明度（LV_OPA_30）来模拟原图效果 */
    }

    /* 2. 核心大数字时速显示 */
    lbl_speed_num = lv_label_create(speed_panel);
    lv_obj_add_style(lbl_speed_num, &style_speed_large, 0);
    lv_label_set_text(lbl_speed_num, "0");
    lv_obj_align(lbl_speed_num, LV_ALIGN_CENTER, 0, 15);
}

/**
 * @brief 创建右侧状态面板 (红圈限速牌、链路状态图标)
 */
static void create_right_status_panel(void) {
    status_panel = lv_obj_create(scr);
    lv_obj_add_style(status_panel, &style_panel_transparent, 0);
    lv_obj_set_size(status_panel, 65, 240);
    lv_obj_align(status_panel, LV_ALIGN_RIGHT_MID, -5, 0);

    /* 1. 红圈白底黑字限速牌 */
    lbl_speed_limit = lv_label_create(status_panel);
    lv_obj_add_style(lbl_speed_limit, &style_speed_limit_circle, 0);
    lv_obj_set_size(lbl_speed_limit, 42, 42);
    lv_label_set_text(lbl_speed_limit, "80");
    lv_obj_set_style_text_align(lbl_speed_limit, LV_TEXT_ALIGN_CENTER, 0);
    lv_obj_set_style_pad_top(lbl_speed_limit, 8, 0); /* 垂直居中微调微调，根据字体大小调整 */
    lv_obj_align(lbl_speed_limit, LV_ALIGN_TOP_MID, 0, 45);

    /* 2. 设备连接状态图标（如原图链条图标） */
    img_status_link = lv_label_create(status_panel);
    lv_obj_add_style(img_status_link, &style_text_white, 0);
    lv_label_set_text(img_status_link, "🔗"); /* 符号占位，可填充为精美切图 */
    lv_obj_align(img_status_link, LV_ALIGN_BOTTOM_MID, 0, -50);
}

/**
 * @brief HUD UI 系统的主入口初始化 API
 * @note 请在初始化好显示屏驱动并调用 lv_init() 后调用此函数
 */
void hud_ui_init(void) {
    /* 获取当前活跃的主屏幕句柄 */
    scr = lv_scr_act();
    
    /* 初始化样式表 */
    init_ui_styles();
    lv_obj_add_style(scr, &style_scr, 0);

    /* 顺序构建左、中、右三大板块 */
    create_left_navigation_panel();
    create_center_speed_panel();
    create_right_status_panel();
}

/**
 * @brief 动态更新当前车速
 * @param speed 当前速度值 (km/h)
 */
void hud_ui_update_speed(int speed) {
    if(lbl_speed_num) {
        lv_label_set_text_fmt(lbl_speed_num, "%d", speed);
    }
}

/**
 * @brief 动态更新左侧导航路名与距离
 * @param road_name 道路名称字符串 (UTF-8 编码)
 * @param distance_meters 剩余距离 (米)
 */
void hud_ui_update_navigation(const char * road_name, int distance_meters) {
    if(lbl_road_name) {
        lv_label_set_text(lbl_road_name, road_name);
    }
    if(lbl_distance) {
        lv_label_set_text_fmt(lbl_distance, "%d米", distance_meters);
    }
}

/**
 * @brief 动态更新限速值
 * @param limit 限速数值 (如 80, 120)
 */
void hud_ui_update_speed_limit(int limit) {
    if(lbl_speed_limit) {
        lv_label_set_text_fmt(lbl_speed_limit, "%d", limit);
    }
}
如何替换资源？
​图片：使用 SquareLine Studio 或 LVGL Image Converter 将你的弯头图标、链条图标转为 .c 数组。然后将 img_nav_arrow = lv_label_create(...) 升级成 lv_img_create()，并调用 lv_img_set_src(img_nav_arrow, &你的图片结构体)。
​汉字与大数字：原图的效果精美核心在于字体。你需要使用 LVGL 的字体转换器，把思源黑体或车载黑体转换为含有 0-9、米、成洛大道 等字样的位图字体文件，然后在代码顶部用 LV_FONT_DECLARE(你的字体名) 引入，并通过 lv_style_set_text_font() 分别赋给大数字和路名标签。
如何使车道指示器动起来？
​lane_container 内部使用的是标准的 Flexbox（弹性盒子）布局。你可以为内部的每一个小车道控件建立句柄，当导航 SDK 传回车道数据时，通过 lv_obj_set_style_opa(lane_arrow, LV_OPA_30, 0) 将不需要或不能走的车道变暗，高亮可以走的车道。
主程序如何调用？
在你的主程序（如 main.cpp 或 main.c）中，确保执行顺序如下：
// 1. 初始化屏幕和LVGL内核
lv_init();
tft_lvgl_driver_init(); // 你的屏幕驱动初始化

// 2. 初始化此 UI 框架
hud_ui_init();

// 3. 在主循环或定时器中更新数据
while(1) {
    lv_timer_handler(); // 处理 LVGL 任务

    // 模拟从地图/GPS接收数据并更新
    hud_ui_update_speed(current_gps_speed); 
    usleep(5000);
}
